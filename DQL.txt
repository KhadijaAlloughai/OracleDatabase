-- From → Join → where → Group by → having → select → order by 
--1-Retrieve the first and last name of all employees along with their gender and birth date.
SELECT 
    FName AS First_Name, 
    LName AS Last_Name, 
    Gender,
    BD AS Birth_Date
FROM Employee;

--2-List all employees who work in a specific department.
-- From → Join → where → Group by → having → select → order by 
SELECT 
    SSN, 
    FName || ' ' || LName AS Employee_Name, 
    Deparnumber AS Department_ID
FROM Employee
WHERE Deparnumber = (
    SELECT Deparnumber 
    FROM Department 
    WHERE DeparName = 'HR Department');
    
    select * from department;
    select * from employee;
    
--3-Display all projects along with their location and city.
SELECT 
    PName AS Project_Name, 
    Location, 
    City
FROM Project;

--4-Find the employee who manages each department along with their hire date.
-- From → Join → where → Group by → having → select → order by 
SELECT 
    d.DeparNumber AS Department_ID,
    d.DeparName AS Department_Name,
    e.FName || ' ' || e.LName AS Manager_Name,
    d.HiringDate AS Manager_Hire_Date
FROM Department d
INNER JOIN Employee e ON d.SSN = e.SSN;

--5-Retrieve all dependents associated with a specific employee.
SELECT 
    e.SSN AS Employee_SSN,
    e.FName || ' ' || e.LName AS Employee_Name,
    dp.DepName AS Dependent_Name,
    dp.BD AS Dependent_Birth_Date,
    dp.Gender AS Dependent_Gender
FROM Employee e
INNER JOIN Dependent dp ON e.SSN = dp.SSN
WHERE e.SSN = 100;

--6-List all employees who work on a specific project along with their working hours.
SELECT 
    e.SSN AS Employee_SSN,
    e.FName || ' ' || e.LName AS Employee_Name,
    p.PName AS Project_Name,
    w.Hours AS Working_Hours
FROM Employee e
INNER JOIN WorkingHours w ON e.SSN = w.SSN
INNER JOIN Project p  ON w.PNumber = p.PNumber   
WHERE p.PName = 'Cloud System';

select * from project;

--8-Find all employees supervised by a specific employee.
SELECT 
    e.SSN AS Employee_SSN,
    e.FName || ' ' || e.LName AS Employee_Name,
    m.SSN AS Supervisor_SSN,
    m.FName || ' ' || m.LName AS Supervisor_Name
FROM Employee e
LEFT JOIN Employee m ON e.SUPERVISORID = m.SSN
ORDER BY e.SSN;
